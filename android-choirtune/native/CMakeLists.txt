cmake_minimum_required(VERSION 3.18)
project(choirdemucs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DEPS "${ROOT}/.cache/demucs-native")
file(GLOB INFERENCE "${DEPS}/cpp/src/*.cpp")
# A single-threaded bounded inference job: no OpenMP/BLAS/native downloads.
add_library(demucs_engine STATIC ${INFERENCE} engine.cpp streaming.cpp)
set_target_properties(demucs_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(demucs_engine PUBLIC "${DEPS}/cpp/src" "${DEPS}/eigen" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(demucs_engine PUBLIC EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
target_compile_options(demucs_engine PRIVATE -O2 -DNDEBUG)
if(ANDROID)
  add_library(choirdemucs SHARED jni.cpp)
  target_link_libraries(choirdemucs PRIVATE demucs_engine log)
  target_link_options(choirdemucs PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(streaming_test test_streaming.cpp streaming.cpp)
  target_compile_options(streaming_test PRIVATE -O2)
  add_executable(demucs_smoke smoke.cpp)
  target_link_libraries(demucs_smoke PRIVATE demucs_engine)
endif()
